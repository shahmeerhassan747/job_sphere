============================================================

-- Create Table
CREATE TABLE jobs (
    id          SERIAL UNIQUE PRIMARY KEY,
    title       VARCHAR(500),
    description TEXT,

    type        VARCHAR(100),
    company_id  INT REFERENCES companies(id),
    posted_by   INT REFERENCES users(id),

4. SALARIES TABLE
============================================================

-- Create Table
CREATE TABLE salaries (
    id          SERIAL UNIQUE PRIMARY KEY,
    min_salary  INT,
    max_salary  INT,
    currency    VARCHAR(10),
    location    VARCHAR(500),
    job_id      INT REFERENCES jobs(id),

);

5. JOIN QUERIES
============================================================

-- 1. Get All Jobs with Company and Salary
SELECT j.id, j.title, j.location, j.type,
       c.name AS company_name, c.logo AS company_logo,
       s.min_salary, s.max_salary, s.currency
FROM jobs j
JOIN companies c ON j.company_id = c.id
LEFT JOIN salaries s ON s.job_id = j.id;

-- 2. Get Specific Job Details
SELECT j.id, j.title, j.description, j.location, j.type,
       c.name AS company_name, c.logo, c.website, c.industry,
       s.min_salary, s.max_salary, s.currency
FROM jobs j
JOIN companies c ON j.company_id = c.id
LEFT JOIN salaries s ON s.job_id = j.id
WHERE j.id = 6;

-- 3. Get All Companies with Total Jobs
SELECT c.id, c.name, c.industry, c.location,
       COUNT(j.id) AS total_jobs
FROM companies c
LEFT JOIN jobs j ON j.company_id = c.id
GROUP BY c.id, c.name, c.industry, c.location;

-- 4. Get All Salaries with Job and Company
SELECT s.id, s.min_salary, s.max_salary, s.currency,
       j.title AS job_title, j.location,
       c.name  AS company_name
FROM salaries s
JOIN jobs j ON s.job_id = j.id
JOIN companies c ON j.company_id = c.id;

-- 5. Search Jobs by Title
SELECT j.id, j.title, j.location, j.type,
       c.name AS company_name,
       s.min_salary, s.max_salary
FROM jobs j
JOIN companies c ON j.company_id = c.id
LEFT JOIN salaries s ON s.job_id = j.id
WHERE j.title ILIKE '%developer%';

-- 6. Search Jobs by Location
SELECT j.id, j.title, j.location, j.type,
       c.name AS company_name,
       s.min_salary, s.max_salary
FROM jobs j
JOIN companies c ON j.company_id = c.id
LEFT JOIN salaries s ON s.job_id = j.id
WHERE j.location ILIKE '%remote%';

-- 7. Get Jobs Posted by Specific User
SELECT j.id, j.title, j.location, j.type,
       c.name AS company_name,
       u.name AS posted_by
FROM jobs j
JOIN companies c ON j.company_id = c.id
JOIN users u ON j.posted_by = u.id
WHERE u.id = 2;
